```js
import { test } from "node:test";
import assert from "node:assert/strict";
import { paginate } from "../src/paginate.js";

const ITEMS = [1, 2, 3, 4, 5, 6, 7];

test("page 1 is the first chunk", () => {
  const r = paginate(ITEMS, 1, 3);
  assert.deepEqual(r.items, [1, 2, 3], "page 1 should return the first 3 items");
  assert.equal(r.page, 1, "page should be echoed back as 1");
});

test("page 2 is the second chunk", () => {
  const r = paginate(ITEMS, 2, 3);
  assert.deepEqual(r.items, [4, 5, 6], "page 2 should return items 4-6");
  assert.equal(r.page, 2, "page should be echoed back as 2");
});

test("last partly full page returns the remaining items", () => {
  const r = paginate(ITEMS, 3, 3);
  assert.deepEqual(r.items, [7], "page 3 should return the single leftover item");
});

test("totalPages counts a partly full last page", () => {
  assert.equal(paginate(ITEMS, 1, 3).totalPages, 3, "7 items at size 3 should be 3 pages");
  assert.equal(paginate([1, 2, 3, 4, 5], 1, 2).totalPages, 3, "5 items at size 2 should be 3 pages");
});

test("totalPages for an exact multiple", () => {
  assert.equal(paginate([1, 2, 3, 4, 5, 6], 1, 2).totalPages, 3, "6 items at size 2 should be 3 pages");
});

test("a page past the end gives an empty page instead of throwing", () => {
  let r;
  assert.doesNotThrow(() => { r = paginate(ITEMS, 10, 3); }, "page past the end should not throw");
  assert.deepEqual(r.items, [], "page past the end should have no items");
  assert.equal(r.totalPages, 3, "totalPages should still be 3 for a page past the end");
});

test("the page right after the last is empty", () => {
  const r = paginate(ITEMS, 4, 3);
  assert.deepEqual(r.items, [], "page 4 of 3 should be empty");
});

test("size less than 1 throws RangeError", () => {
  assert.throws(() => paginate(ITEMS, 1, 0), RangeError, "size 0 should throw RangeError");
  assert.throws(() => paginate(ITEMS, 1, -2), RangeError, "negative size should throw RangeError");
});

test("page less than 1 throws RangeError", () => {
  assert.throws(() => paginate(ITEMS, 0, 3), RangeError, "page 0 should throw RangeError");
  assert.throws(() => paginate(ITEMS, -1, 3), RangeError, "negative page should throw RangeError");
});

test("size 1 boundary is valid", () => {
  const r = paginate([9, 8, 7], 2, 1);
  assert.deepEqual(r.items, [8], "size 1 page 2 should give the second item");
  assert.equal(r.totalPages, 3, "3 items at size 1 should be 3 pages");
});

test("size larger than the list gives everything on page 1", () => {
  const r = paginate([1, 2, 3], 1, 10);
  assert.deepEqual(r.items, [1, 2, 3], "all items should fit on page 1");
  assert.equal(r.totalPages, 1, "should be a single page");
});

test("does not modify the input array", () => {
  const input = [1, 2, 3, 4];
  paginate(input, 1, 2);
  assert.deepEqual(input, [1, 2, 3, 4], "input array should be unchanged");
});
```
