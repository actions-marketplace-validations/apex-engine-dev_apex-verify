```js
import { test } from "node:test";
import assert from "node:assert/strict";
import { LRUCache } from "../src/lru.js";

test("reading a key protects it from eviction", () => {
  const c = new LRUCache(2);
  c.set("a", 1); c.set("b", 2);
  c.get("a");
  c.set("c", 3);
  assert.equal(c.get("a"), 1, "a was read recently and should stay");
  assert.equal(c.get("b"), undefined, "b is least recently used and should be evicted");
  assert.equal(c.get("c"), 3, "c should be present");
});

test("re-setting an existing key in a full cache evicts nothing", () => {
  const c = new LRUCache(2);
  c.set("a", 1); c.set("b", 2);
  c.set("a", 10);
  assert.equal(c.size, 2, "size should stay at 2");
  assert.equal(c.get("a"), 10, "a should have the new value");
  assert.equal(c.get("b"), 2, "b should not be evicted by re-setting a");
});

test("re-setting a key makes it most recently used", () => {
  const c = new LRUCache(2);
  c.set("a", 1); c.set("b", 2);
  c.set("a", 10);
  c.set("c", 3);
  assert.equal(c.get("b"), undefined, "b is now least recently used and should be evicted");
  assert.equal(c.get("a"), 10, "a should remain after being re-set");
  assert.equal(c.get("c"), 3, "c should be present");
});

test("least recently used order follows reads across several keys", () => {
  const c = new LRUCache(3);
  c.set("a", 1); c.set("b", 2); c.set("c", 3);
  c.get("a"); c.get("b");
  c.set("d", 4);
  assert.equal(c.get("c"), undefined, "c is least recently used and should be evicted");
  assert.equal(c.get("a"), 1, "a should remain");
  assert.equal(c.get("b"), 2, "b should remain");
  assert.equal(c.get("d"), 4, "d should be present");
});

test("size never exceeds capacity", () => {
  const c = new LRUCache(3);
  for (let i = 0; i < 10; i++) c.set("k" + i, i);
  assert.equal(c.size, 3, "size should equal capacity after many inserts");
});

test("getting a missing key returns undefined and does not change size", () => {
  const c = new LRUCache(2);
  c.set("a", 1);
  assert.equal(c.get("zzz"), undefined, "missing key should give undefined");
  assert.equal(c.size, 1, "size should be unchanged");
});

test("oldest entry is evicted when nothing was read", () => {
  const c = new LRUCache(2);
  c.set("a", 1); c.set("b", 2); c.set("c", 3);
  assert.equal(c.get("a"), undefined, "a is oldest and should be evicted");
  assert.equal(c.get("b"), 2, "b should remain");
  assert.equal(c.get("c"), 3, "c should remain");
});
```
