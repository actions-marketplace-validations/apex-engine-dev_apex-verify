```js
import { test } from "node:test";
import assert from "node:assert/strict";
import { daysBetween } from "../src/dates.js";

test("crosses March DST transition (Spring Forward)", () => {
  assert.equal(
    daysBetween("2024-03-09", "2024-03-11"),
    2,
    "Should be exactly 2 days across Spring DST transition"
  );
});

test("crosses November DST transition (Fall Back)", () => {
  assert.equal(
    daysBetween("2024-11-02", "2024-11-04"),
    2,
    "Should be exactly 2 days across Fall DST transition"
  );
});

test("same day", () => {
  assert.equal(
    daysBetween("2024-05-05", "2024-05-05"),
    0,
    "Same day should be 0 days"
  );
});

test("negative days (b before a)", () => {
  assert.equal(
    daysBetween("2024-03-11", "2024-03-09"),
    -2,
    "Reversed dates should yield negative days"
  );
});

test("leap year crossing", () => {
  assert.equal(
    daysBetween("2024-02-28", "2024-03-01"),
    2,
    "2024 is a leap year, should be 2 days"
  );
});

test("non-leap year crossing", () => {
  assert.equal(
    daysBetween("2023-02-28", "2023-03-01"),
    1,
    "2023 is not a leap year, should be 1 day"
  );
});

test("preserves existing simple behavior", () => {
  assert.equal(
    daysBetween("2024-01-01", "2024-01-31"),
    30,
    "Simple month difference should remain 30 days"
  );
});
```
