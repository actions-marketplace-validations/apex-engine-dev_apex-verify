```js
import { test } from "node:test";
import assert from "node:assert/strict";
import { parseCSV } from "../src/csv.js";

test("handles commas inside quotes", () => {
  assert.deepEqual(
    parseCSV('x,"Smith, John",y\n'),
    [["x", "Smith, John", "y"]],
    "Should not split at commas inside quotes"
  );
});

test("handles doubled quotes inside quotes", () => {
  assert.deepEqual(
    parseCSV('a,"b ""c"" d",e\n'),
    [["a", 'b "c" d', "e"]],
    "Should unescape doubled quotes"
  );
});

test("handles line breaks inside quotes", () => {
  assert.deepEqual(
    parseCSV('a,"b\nc",d\n'),
    [["a", "b\nc", "d"]],
    "Should not split rows at newlines inside quotes"
  );
});

test("handles mixed quoting, commas, and newlines", () => {
  assert.deepEqual(
    parseCSV('1,"Smith, John","Line1\nLine2","He said ""Hello"""\n'),
    [["1", "Smith, John", "Line1\nLine2", 'He said "Hello"']],
    "Should handle all quoting rules in one row"
  );
});

test("handles empty fields and quoted empty fields", () => {
  assert.deepEqual(
    parseCSV('a,,c\n1,"",3\n'),
    [["a", "", "c"], ["1", "", "3"]],
    "Should parse empty fields correctly"
  );
});

test("handles empty input", () => {
  assert.deepEqual(
    parseCSV(""),
    [],
    "Empty string should return empty array"
  );
});

test("plain csv without quotes", () => {
  assert.deepEqual(
    parseCSV("a,b\n1,2\n"),
    [["a", "b"], ["1", "2"]],
    "Should parse plain CSV correctly"
  );
});
```
